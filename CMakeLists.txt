cmake_minimum_required(VERSION 3.24)
project(sPDFViewer VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "sPDF requires a Windows x64 toolchain (MSVC or MinGW-w64).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(cmake/Pdfium.cmake)
add_library(spdf_core STATIC src/PdfiumLibrary.cpp src/PdfDocument.cpp src/PdfRenderer.cpp src/TabManager.cpp)
target_include_directories(spdf_core PUBLIC include "${SPDF_PDFIUM_DIR}/include")
target_compile_definitions(spdf_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
add_library(spdf_ui STATIC src/MainWindow.cpp src/DocumentTabBar.cpp)
target_link_libraries(spdf_ui PUBLIC spdf_core comctl32 comdlg32 shell32 shlwapi gdi32 user32 uxtheme)
target_include_directories(spdf_ui PRIVATE resources)
add_executable(sPDF WIN32 src/main.cpp resources/app.rc)
target_link_libraries(sPDF PRIVATE spdf_ui)
target_include_directories(sPDF PRIVATE resources)
if(MSVC)
  target_compile_options(spdf_core PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(spdf_ui PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(sPDF PRIVATE /W4 /utf-8 /permissive-)
  target_link_options(sPDF PRIVATE /MANIFEST:NO)
else()
  target_compile_options(spdf_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(spdf_ui PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(sPDF PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(sPDF PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
add_custom_command(TARGET sPDF POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${SPDF_PDFIUM_DIR}/bin/pdfium.dll" "$<TARGET_FILE_DIR:sPDF>/pdfium.dll")
include(CTest)
if(BUILD_TESTING)
  add_executable(spdf_pdf_tests tests/PdfIntegrationTests.cpp)
  target_link_libraries(spdf_pdf_tests PRIVATE spdf_core)
  if(MSVC)
    target_compile_options(spdf_pdf_tests PRIVATE /W4 /utf-8)
  else()
    target_link_options(spdf_pdf_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET spdf_pdf_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${SPDF_PDFIUM_DIR}/bin/pdfium.dll" "$<TARGET_FILE_DIR:spdf_pdf_tests>/pdfium.dll")
  add_test(NAME pdfium_integration COMMAND spdf_pdf_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_executable(spdf_tab_tests tests/TabIntegrationTests.cpp)
  target_link_libraries(spdf_tab_tests PRIVATE spdf_core)
  if(MSVC)
    target_compile_options(spdf_tab_tests PRIVATE /W4 /utf-8)
  else()
    target_link_options(spdf_tab_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET spdf_tab_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${SPDF_PDFIUM_DIR}/bin/pdfium.dll" "$<TARGET_FILE_DIR:spdf_tab_tests>/pdfium.dll")
  add_test(NAME document_tabs COMMAND spdf_tab_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(pdfium_integration document_tabs PROPERTIES TIMEOUT 90)
  add_executable(spdf_window_tests tests/WindowIntegrationTests.cpp resources/app.rc)
  target_include_directories(spdf_window_tests PRIVATE resources)
  target_link_libraries(spdf_window_tests PRIVATE spdf_ui)
  if(MSVC)
    target_compile_options(spdf_window_tests PRIVATE /W4 /utf-8)
    target_link_options(spdf_window_tests PRIVATE /MANIFEST:NO)
  else()
    target_link_options(spdf_window_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME window_tabs COMMAND spdf_window_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(window_tabs PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()
install(TARGETS sPDF RUNTIME DESTINATION sPDF)
install(FILES "${SPDF_PDFIUM_DIR}/bin/pdfium.dll" DESTINATION sPDF)
install(FILES "${SPDF_PDFIUM_DIR}/LICENSE" DESTINATION sPDF/licenses/pdfium)
install(FILES docs/PDFium.md DESTINATION sPDF/licenses/pdfium)
install(DIRECTORY resources/icons/ DESTINATION sPDF/resources/icons)
install(DIRECTORY LICENSES/ DESTINATION sPDF/licenses)
install(FILES docs/README.txt DESTINATION sPDF)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "sPDF-${PROJECT_VERSION}-preview-win-x64")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
include(CPack)
